// Mescle com as regras atuais do projeto e PUBLIQUE no Firebase Console.
// Não substitua regras existentes sem revisar.

function signedIn() {
  return request.auth != null;
}

match /public_factories/{uid} {
  allow read: if signedIn();
  allow create, update: if signedIn()
    && request.auth.uid == uid
    && request.resource.data.uid == uid;
  allow delete: if signedIn() && request.auth.uid == uid;
}

// Compatibilidade somente para recuperar o próprio perfil legado.
match /players/{uid} {
  allow get: if signedIn() && request.auth.uid == uid;
}

// Compatibilidade para a coleção de conta, se ela existir no projeto.
match /player_accounts/{uid} {
  allow get: if signedIn() && request.auth.uid == uid;
}

match /character_offers/{ownerId} {
  allow read: if signedIn();
  allow create, delete: if signedIn() && request.auth.uid == ownerId;
  allow update: if signedIn() && (
    request.auth.uid == ownerId ||
    (
      request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly(['leasedBy', 'leasedUntil']) &&
      request.resource.data.leasedBy == request.auth.uid &&
      resource.data.leasedUntil <= request.time &&
      request.resource.data.leasedUntil > request.time
    )
  );
}

match /character_rentals/{rentalId} {
  allow create: if signedIn()
    && request.resource.data.renterUid == request.auth.uid;
  allow read: if signedIn()
    && (
      resource.data.renterUid == request.auth.uid ||
      resource.data.ownerUid == request.auth.uid
    );
  allow update, delete: if false;
}
